function doGet(e) { var ss = SpreadsheetApp.openById("10AZuV1hdJfdwfUCrJJi9Bhh4Cb0jqZh106RPowBxzQk"); var view = e.parameter.view; var uid = e.parameter.userId; var out = {}; var cust = ss.getSheetByName("ลูกค้า").getDataRange().getValues(); var found = false, custName, custPoints; for (var i = 1; i < cust.length; i++) { if (cust[i][0] === uid) { found = true; custName = cust[i][1]; custPoints = cust[i][3]; break; } } out.found = found; if (!found) { return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON); } if (view === "reward") { out.name = custName; out.points = custPoints; var hist = ss.getSheetByName("ประวัติแต้ม").getDataRange().getValues(); out.history = hist.filter(function(r, i){ return i > 0 && r[0] === uid; }) .map(function(r){ return {date:r[1], earned:r[2], used:r[3], reward:r[4]}; }); } else if (view === "sales") { var s = ss.getSheetByName("ยอดขาย").getDataRange().getValues(); var rows = s.filter(function(r, i){ return i > 0 && r[0] === uid; }); out.records = rows.map(function(r){ return {date:r[1], weight:r[2], amount:r[3], receipt:r[4]}; }); out.totalAmount = rows.reduce(function(sum,r){ return sum + Number(r[3]||0); }, 0); out.totalWeight = rows.reduce(function(sum,r){ return sum + Number(r[2]||0); }, 0); out.count = rows.length; } return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var ss = SpreadsheetApp.openById("10AZuV1hdJfdwfUCrJJi9Bhh4Cb0jqZh106RPowBxzQk"); var data = JSON.parse(e.postData.contents); if (data.action === "register") { var sheet = ss.getSheetByName("ลูกค้า"); sheet.appendRow([data.userId, data.name, data.phone, 0]); return ContentService.createTextOutput(JSON.stringify({success:true})).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({success:false})).setMimeType(ContentService.MimeType.JSON); }
